select priezvisko,plat from zamestnanec
where plat> (select avg(plat) from zamestnanec);

select meno, priezvisko from zamestnanec
where pracuje_na =
(select pracuje_na from zamestnanec
where priezvisko = 'Novák')
and priezvisko != 'Novák';

select * from zamestnanec
where pracuje_na in(
select pracuje_na from zamestnanec
where meno in ('Pavol','Jozef'));

select priezvisko,datum_narodenia from zamestnanec
where datum_narodenia =
(select min(datum_narodenia) from zamestnanec)
or datum_narodenia =
(select max(datum_narodenia) from zamestnanec);

select upper(priezvisko), lower(meno),length(meno) from zamestnanec
where lower(priezvisko) like '%b%';

select meno,priezvisko,length(priezvisko) from zamestnanec
where length(priezvisko)>6
order by length(priezvisko) desc;

select count(*),length(meno) from zamestnanec
where length(meno)  between 4 and 6
group by length(meno)
order by count(*) asc;

select count(*),length(meno) from zamestnanec
group by length(meno)
having length(meno) between 4 and 6
order by count(*) asc;

update zamestnanec set plat = 100 where plat = 0;

select plat, 1000/plat,round(1000/plat,3), trunc(1000/plat,3), round(plat,-2) from zamestnanec
order by plat asc;

from dual;
